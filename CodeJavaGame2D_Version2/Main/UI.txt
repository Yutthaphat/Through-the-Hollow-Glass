package JavaLearn; // บอกว่าคลาสนี้อยู่ในแพ็กเกจชื่อ JavaLearn

import java.awt.Color; // import คลาส Color สำหรับกำหนดสี
import java.awt.Font; // import คลาส Font สำหรับกำหนดฟอนต์
import java.awt.Graphics2D; // import คลาส Graphics2D สำหรับการวาดภาพ 2 มิติ
import java.awt.image.BufferedImage; // import คลาส BufferedImage สำหรับจัดการรูปภาพ
import java.text.DecimalFormat; // import DecimalFormat สำหรับจัดรูปแบบตัวเลขทศนิยม

import Object.OBJ_Key; // import คลาส OBJ_Key

public class UI { // คลาสสำหรับส่วนติดต่อผู้ใช้ (User Interface)

	GamePanel gp; // ตัวแปรอ้างอิงถึง GamePanel เพื่อเข้าถึงข้อมูลเกม
	Font arial_40, arial_80B; // ตัวแปร Font สำหรับฟอนต์ขนาด 40 (ธรรมดา) และ 80 (ตัวหนา)
	BufferedImage keyImage; // ตัวแปรสำหรับเก็บรูปภาพกุญแจ
	public boolean messageOn = false; // สถานะการแสดงข้อความ (true = กำลังแสดง, false = ไม่แสดง)
	public String message = ""; // ข้อความที่จะแสดง
	int messageCounter = 0; // ตัวนับสำหรับควบคุมระยะเวลาการแสดงข้อความ
	public boolean gameFinished = false; // สถานะเกมจบแล้ว (true = จบแล้ว, false = ยังไม่จบ)
	
	double playTime; // เวลาที่ผู้เล่นเล่นเกม
	DecimalFormat dFormat = new DecimalFormat("#0.00"); // Object สำหรับจัดรูปแบบทศนิยมให้มี 2 ตำแหน่ง

	public UI(GamePanel gp) { // Constructor ของ UI
		this.gp = gp; // กำหนด GamePanel ที่รับมาให้กับตัวแปร gp ของคลาสนี้
		
		arial_40 = new Font("Arial", Font.PLAIN, 40); // สร้างฟอนต์ Arial ขนาด 40 แบบธรรมดา
		arial_80B = new Font("Arial", Font.BOLD, 80); // สร้างฟอนต์ Arial ขนาด 80 แบบตัวหนา
		OBJ_Key key = new OBJ_Key(gp); // สร้าง Object กุญแจชั่วคราวเพื่อดึงรูปภาพมาใช้ใน UI
		keyImage = key.image; // เก็บรูปภาพกุญแจ
	}
	
	public void showMessage(String text) { // เมธอดสำหรับแสดงข้อความบนหน้าจอ
		message = text; // กำหนดข้อความที่จะแสดง
		messageOn = true; // ตั้งค่าสถานะให้แสดงข้อความ
	}
	
	public void draw(Graphics2D g2) { // เมธอดสำหรับวาด UI บนหน้าจอ
		
		if(gameFinished == true) { // ถ้าเกมจบแล้ว
			
			g2.setFont(arial_40); // กำหนดฟอนต์เป็น Arial 40
			g2.setColor(Color.white); // กำหนดสีตัวอักษรเป็นสีขาว
			
			String text; // ตัวแปรสำหรับเก็บข้อความ
			int textLength; // ตัวแปรสำหรับเก็บความยาวของข้อความ (เป็นพิกเซล)
			int x; // ตำแหน่ง X สำหรับวาดข้อความ
			int y; // ตำแหน่ง Y สำหรับวาดข้อความ
			
			text = "You found the treasure!"; // ข้อความหลัก
			textLength = (int)g2.getFontMetrics().getStringBounds(text, g2).getWidth(); // คำนวณความยาวข้อความ
			x = gp.screenWidth/2 - textLength/2; // คำนวณตำแหน่ง X เพื่อให้ข้อความอยู่กึ่งกลางหน้าจอ
			y = gp.screenHeight/2 - (gp.tileSize*3); // กำหนดตำแหน่ง Y
			g2.drawString(text, x, y); // วาดข้อความ
			
			text = "Your Time is :" + dFormat.format(playTime) + "!"; // ข้อความแสดงเวลาที่เล่น
			textLength = (int)g2.getFontMetrics().getStringBounds(text, g2).getWidth();
			x = gp.screenWidth/2 - textLength/2;
			y = gp.screenHeight/2 + (gp.tileSize*4);
			g2.drawString(text, x, y);
			
			g2.setFont(arial_80B); // กำหนดฟอนต์เป็น Arial 80 ตัวหนา
			g2.setColor(Color.yellow); // กำหนดสีตัวอักษรเป็นสีเหลือง
			text = "Congratulations!"; // ข้อความแสดงความยินดี
			textLength = (int)g2.getFontMetrics().getStringBounds(text, g2).getWidth();
			x = gp.screenWidth/2 - textLength/2;
			y = gp.screenHeight/2 + (gp.tileSize*2);
			g2.drawString(text, x, y);
			
			gp.gameThread = null; // ตั้งค่า Game Thread ให้เป็น null เพื่อหยุดเกม
		}
		else { // ถ้าเกมยังไม่จบ
			g2.setFont(arial_40); // กำหนดฟอนต์เป็น Arial 40
			g2.setColor(Color.white); // กำหนดสีตัวอักษรเป็นสีขาว
			g2.drawImage(keyImage, gp.tileSize/2, gp.tileSize/2, gp.tileSize, gp.tileSize, null); // วาดรูปกุญแจที่มุมซ้ายบน
			g2.drawString("x " + gp.player.hasKey, 74, 65); // วาดจำนวนกุญแจที่ผู้เล่นมี
			
			// วาดเวลาที่เล่น
			playTime += (double)1/60; // เพิ่มเวลาเล่นทุกเฟรม (1/60 วินาทีต่อเฟรมที่ 60 FPS)
			g2.drawString("Time : " + dFormat.format(playTime), gp.tileSize*11, 65); // วาดข้อความเวลา
			
			// ส่วน MESSAGE (ข้อความ Pop-up)
			if(messageOn == true) { // ถ้ามีข้อความกำลังแสดงอยู่
				
				g2.setFont(g2.getFont().deriveFont(30F)); // ปรับขนาดฟอนต์ของข้อความเป็น 30
				g2.drawString(message, gp.tileSize/2, gp.tileSize*5); // วาดข้อความ
				
				messageCounter++; // เพิ่มตัวนับเวลาแสดงข้อความ
				
				if(messageCounter > 120) { // ถ้าแสดงเกิน 120 เฟรม (ประมาณ 2 วินาที)
					messageCounter = 0; // รีเซ็ตตัวนับ
					messageOn = false; // ปิดการแสดงข้อความ
				}
			}
		}
	}
}
