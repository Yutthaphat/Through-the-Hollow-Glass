package Object; // บอกว่าคลาสนี้อยู่ในแพ็กเกจชื่อ Object

import java.io.IOException; // import IOException สำหรับจัดการข้อผิดพลาดเกี่ยวกับการอ่าน/เขียนไฟล์

import javax.imageio.ImageIO; // import ImageIO สำหรับอ่านไฟล์รูปภาพ

import JavaLearn.GamePanel; // import GamePanel เพื่อเข้าถึงข้อมูลเกม

public class OBJ_Door extends SuperObject { // คลาส OBJ_Door สืบทอดจาก SuperObject
	
	GamePanel gp; // ตัวแปรอ้างอิงถึง GamePanel
	
	public OBJ_Door(GamePanel gp){ // Constructor ของ OBJ_Door
		
		this.gp = gp; // กำหนด GamePanel ที่รับมา
			
		name = "Door"; // กำหนดชื่อของ Object เป็น "Door"
		try {
			// โหลดรูปภาพประตูจาก Resource
			image = ImageIO.read(getClass().getResourceAsStream("/objects/door.png"));
			// ปรับขนาดรูปภาพ
			image = uTool.scaleImage(image, gp.tileSize, gp.tileSize);
			
		}catch(IOException e) { // ดักจับข้อผิดพลาด
			e.printStackTrace();
		}
		collision = true; // ตั้งค่าให้ประตูมีการชน (ผู้เล่นไม่สามารถเดินผ่านได้)
	}
}
