	public void setup(int index, String imageName, boolean collision) { // เมธอดช่วยโหลดและปรับขนาดรูปภาพ Tile
		
		UtilityTool uTool = new UtilityTool(); // สร้าง Object ของ UtilityTool
		
		try {
			tile[index] = new Tile(); // สร้าง Object Tile ที่ index ที่กำหนด
			// อ่านรูปภาพ Tile จาก Resource (โฟลเดอร์ /tiles/) ตามชื่อที่กำหนด
			tile[index].image = ImageIO.read(getClass().getResourceAsStream("/tiles/" + imageName + ".png"));
			// ปรับขนาดรูปภาพ Tile ให้เท่ากับ tileSize ของเกม
			tile[index].image = uTool.scaleImage(tile[index].image, gp.tileSize, gp.tileSize);
			tile[index].collision = collision; // กำหนดสถานะ collision ของ Tile
			
		}catch(IOException e) { // ดักจับข้อผิดพลาดหากอ่านไฟล์รูปภาพไม่ได้
			e.printStackTrace(); // พิมพ์ข้อผิดพลาดใน Console
		}
	}

	public void loadMap(String filePath) { // เมธอดสำหรับโหลดข้อมูลแผนที่จากไฟล์ .txt
		
		try {
			InputStream is = getClass().getResourceAsStream(filePath); // สร้าง InputStream เพื่ออ่านไฟล์จาก Resource
			BufferedReader br = new BufferedReader(new InputStreamReader(is)); // สร้าง BufferedReader สำหรับอ่านไฟล์ทีละบรรทัด
			
			int col = 0; // ตัวนับคอลัมน์
			int row = 0; // ตัวนับแถว
			// วนลูปอ่านข้อมูลแผนที่จนกว่าจะครบขนาด World Map
			while(col < gp.maxWorldCol && row < gp.maxWorldRow) { 
				
				String line = br.readLine(); // อ่านข้อมูลหนึ่งบรรทัดจากไฟล์
				
				while(col < gp.maxWorldCol) { // วนลูปอ่านข้อมูลในแต่ละคอลัมน์ของบรรทัดนั้น
					
					String numbers[] = line.split(" "); // แยกตัวเลขในบรรทัดด้วยช่องว่าง (" ")
					
					int num = Integer.parseInt(numbers[col]); // แปลง String ที่ได้ให้เป็น Integer
					
					mapTileNum[col][row] = num; // เก็บหมายเลข Tile ลงในอาร์เรย์ mapTileNum
					col++; // เลื่อนไปยังคอลัมน์ถัดไป
				}
				if(col == gp.maxWorldCol) { // ถ้าอ่านครบทุกคอลัมน์ในแถวนั้นแล้ว
					col = 0; // รีเซ็ตคอลัมน์เป็น 0
					row++; // เลื่อนไปยังแถวถัดไป
				}
			}
			br.close(); // ปิด BufferedReader
			
		}catch(Exception e) { // ดักจับข้อผิดพลาดหากมีปัญหาในการโหลดแผนที่
			 // e.printStackTrace();  // สามารถ uncomment เพื่อดูข้อผิดพลาดได้
		}
	}
	
	public void draw(Graphics2D g2) { // เมธอดสำหรับวาด Tile ทั้งหมดบนหน้าจอ
	    int worldCol = 0; // ตัวนับคอลัมน์ใน World Map
	    int worldRow = 0; // ตัวนับแถวใน World Map
	    
	    // วนลูปทั่วทั้ง World Map (ตาม maxWorldCol, maxWorldRow)
	    while(worldCol < gp.maxWorldCol && worldRow < gp.maxWorldRow) { 
	        
	        int tileNum = mapTileNum[worldCol][worldRow]; // เอาหมายเลข Tile จากข้อมูลแผนที่
	        
	        int worldX = worldCol * gp.tileSize; // คำนวณพิกัด X ของ Tile นี้ใน World Map
	        int worldY = worldRow * gp.tileSize; // คำนวณพิกัด Y ของ Tile นี้ใน World Map

	        // คำนวณพิกัดของ Tile นี้บนหน้าจอ (สัมพันธ์กับตำแหน่งผู้เล่น เพื่อทำกล้องติดตาม)
	        int screenX = worldX - gp.player.worldX + gp.player.screenX; // คำนวณตำแหน่ง X บนจอ
	        int screenY = worldY - gp.player.worldY + gp.player.screenY; // คำนวณตำแหน่ง Y บนจอ
	        
	        // เพิ่ม Optimization: วาด Tile เฉพาะเมื่อ Tile นั้นอยู่ในขอบเขตการมองเห็นของหน้าจอเท่านั้น
	        // (เผื่อขอบนอกออกไปอีก 1 Tile เพื่อป้องกันขอบดำเวลาผู้เล่นขยับ)
	        if(worldX + gp.tileSize > gp.player.worldX - gp.player.screenX && // ตรวจสอบว่า Tile อยู่ทางขวาของขอบจอซ้าย
	           worldX - gp.tileSize < gp.player.worldX + gp.player.screenX && // ตรวจสอบว่า Tile อยู่ทางซ้ายของขอบจอขวา
	           worldY + gp.tileSize > gp.player.worldY - gp.player.screenY && // ตรวจสอบว่า Tile อยู่ด้านล่างของขอบจอบน
	           worldY - gp.tileSize < gp.player.worldY + gp.player.screenY) { // ตรวจสอบว่า Tile อยู่ด้านบนของขอบจอล่าง
	            
	            g2.drawImage(tile[tileNum].image, screenX, screenY, null); // วาดรูปภาพ Tile บนหน้าจอ (ใช้รูปที่ปรับขนาดมาแล้ว)
	        }

	        worldCol++; // เลื่อนไปยังคอลัมน์ถัดไป
	         
	        if(worldCol == gp.maxWorldCol) { // ถ้าอ่านครบทุกคอลัมน์ในแถวนั้นแล้ว
	            worldCol = 0; // รีเซ็ตคอลัมน์
	            worldRow++; // เลื่อนไปยังแถวถัดไป
	        }
	    }
	}
