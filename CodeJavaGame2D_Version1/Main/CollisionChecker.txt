// ยังอยู่ในแพ็กเกจ JavaLearn นะ คลาสนี้ไว้เช็คการชนกัน
package JavaLearn;

// ดึงคลาส Entity มาใช้
import Entity.Entity;

// คลาส CollisionChecker นี่แหละคือตำรวจจราจรในเกม คอยเช็คว่าอะไรชนกับอะไรบ้าง
public class CollisionChecker {
	
	GamePanel gp; // อ้างอิงถึง GamePanel เพื่อเข้าถึงข้อมูลเกมต่างๆ
	
	// Constructor ของ CollisionChecker
	public CollisionChecker(GamePanel gp) {
		this.gp = gp;
	}
	
	// เมธอด checkTile ตรวจสอบการชนกันระหว่าง Entity (เช่นผู้เล่น) กับ Tile (พื้น, กำแพง)
	public void checkTile(Entity entity) {
		// คำนวณขอบเขตของพื้นที่ชน (solidArea) ของ Entity ในโลกของเกม
		int entityLeftWorldX = entity.worldX + entity.solidArea.x;
		int entityRightWorldX = entity.worldX + entity.solidArea.x + entity.solidArea.width;
		int entityTopWorldY = entity.worldY + entity.solidArea.y;
		int entityBottomWorldY = entity.worldY + entity.solidArea.y + entity.solidArea.height;
		
		// แปลงพิกัดโลกเป็นพิกัดคอลัมน์/แถวของ Tile
		int entityLeftCol = entityLeftWorldX/gp.tileSize;
		int entityRightCol = entityRightWorldX/gp.tileSize;
		int entityTopRow = entityTopWorldY/gp.tileSize;
		int entityBottomRow = entityBottomWorldY/gp.tileSize;
		
		int tileNum1, tileNum2; // ตัวแปรสำหรับเก็บหมายเลข Tile ที่อาจจะชน
		
		// เช็คการชนตามทิศทางที่ Entity กำลังจะไป
		switch(entity.direction) {
		case "up": // ถ้ากำลังจะเดินขึ้น
			// คำนวณแถวของ Tile ที่จะชน ถ้า Entity เคลื่อนที่ขึ้นด้วยความเร็ว speed
			entityTopRow = (entityTopWorldY - entity.speed)/gp.tileSize;
			// ได้ Tile สองอันที่อยู่ตรงขอบบน
			tileNum1 = gp.tileM.mapTileNum[entityLeftCol][entityTopRow];
			tileNum2 = gp.tileM.mapTileNum[entityRightCol][entityTopRow];
			// ถ้า Tile สองอันนี้อันไหนก็ได้ที่ตั้งค่าชนไว้ (collision == true) ให้บอกว่าชนแล้ว
			if(gp.tileM.tile[tileNum1].collision == true || gp.tileM.tile[tileNum2].collision == true) {
				entity.collisionOn = true;
			}
			break;
		case "down": // ถ้ากำลังจะเดินลง
			entityBottomRow = (entityBottomWorldY + entity.speed)/gp.tileSize;
			tileNum1 = gp.tileM.mapTileNum[entityLeftCol][entityBottomRow];
			tileNum2 = gp.tileM.mapTileNum[entityRightCol][entityBottomRow];
			if(gp.tileM.tile[tileNum1].collision == true || gp.tileM.tile[tileNum2].collision == true) {
				entity.collisionOn = true;
			}
			break;
		case "left": // ถ้ากำลังจะเดินซ้าย
			entityLeftCol = (entityLeftWorldX - entity.speed)/gp.tileSize;
			tileNum1 = gp.tileM.mapTileNum[entityLeftCol][entityTopRow];
			tileNum2 = gp.tileM.mapTileNum[entityLeftCol][entityBottomRow];
			if(gp.tileM.tile[tileNum1].collision == true || gp.tileM.tile[tileNum2].collision == true) {
				entity.collisionOn = true;
			}
			break;
		case "right": // ถ้ากำลังจะเดินขวา
			entityRightCol = (entityRightWorldX + entity.speed)/gp.tileSize;
			tileNum1 = gp.tileM.mapTileNum[entityRightCol][entityTopRow];
			tileNum2 = gp.tileM.mapTileNum[entityRightCol][entityBottomRow];
			if(gp.tileM.tile[tileNum1].collision == true || gp.tileM.tile[tileNum2].collision == true) {
				entity.collisionOn = true;
			}
			break;
		}
	}

	// เมธอด checkObject ตรวจสอบการชนกันระหว่าง Entity กับ Object (กุญแจ ประตู อะไรพวกนี้)
	// player เป็น boolean บอกว่า Entity ที่กำลังเช็คอยู่คือผู้เล่นจริงจังรึเปล่า
	public int checkObject(Entity entity, boolean player) {
		
		int index = 999; // กำหนดค่าเริ่มต้น 999 แปลว่ายังไม่ชน Object อะไรเลย
		
		// วนลูปเช็ค Object ทุกตัวที่มีอยู่ในเกม
		for(int i = 0; i < gp.obj.length; i++) {
			
			// เช็คว่า Object ตัวนั้นมีอยู่จริงมั้ย (ไม่ได้ถูกเก็บไปแล้วหรือเป็น null)
			if(gp.obj[i] != null) {
				
				// --- การปรับตำแหน่ง solidArea ชั่วคราวเพื่อเช็คชน ---
				// ย้ายตำแหน่ง solidArea ของ Entity ให้ตรงกับตำแหน่งจริงในโลก (worldX, worldY)
				entity.solidArea.x = entity.worldX + entity.solidArea.x;
				entity.solidArea.y = entity.worldY + entity.solidArea.y;
				// ทำเหมือนกันกับ Object
				gp.obj[i].solidArea.x = gp.obj[i].worldX + gp.obj[i].solidArea.x;
				gp.obj[i].solidArea.y = gp.obj[i].worldY + gp.obj[i].solidArea.y;
				
				// เช็คการชนตามทิศทางที่ Entity กำลังจะไป
				switch(entity.direction) {
				case "up": // ถ้ากำลังจะเดินขึ้น
					entity.solidArea.y -= entity.speed; // "ลอง" ขยับ solidArea ขึ้นไปก่อน
					// เช็คว่า solidArea ของ Entity มันซ้อนทับกับ solidArea ของ Object รึเปล่า
					if(entity.solidArea.intersects(gp.obj[i].solidArea)) {
						// ถ้า Object นั้นชนได้ (collision == true) ก็ตั้งค่าว่า Entity ชนแล้ว
						if(gp.obj[i].collision == true) {
							entity.collisionOn = true;
						}
						// ถ้า Entity ที่ชนเป็นผู้เล่น ก็คืนค่า index ของ Object นั้นกลับไปให้ Player จัดการ
						if(player == true) {
							index = i;
						}
					}
					break;
				case "down": // ถ้ากำลังจะเดินลง
					entity.solidArea.y += entity.speed;
					if(entity.solidArea.intersects(gp.obj[i].solidArea)) {
						if(gp.obj[i].collision == true) {
							entity.collisionOn = true;
						}
						if(player == true) {
							index = i;
						}
					}
					break;
				case "left": // ถ้ากำลังจะเดินซ้าย
					entity.solidArea.x -= entity.speed;
					if(entity.solidArea.intersects(gp.obj[i].solidArea)) {
						if(gp.obj[i].collision == true) {
							entity.collisionOn = true;
						}
						if(player == true) {
							index = i;
						}
					}
					break;
				case "right": // ถ้ากำลังจะเดินขวา
					entity.solidArea.x += entity.speed;
					if(entity.solidArea.intersects(gp.obj[i].solidArea)) {
						if(gp.obj[i].collision == true) {
							entity.collisionOn = true;
						}
						if(player == true) {
							index = i;
						}
					break; // มี break ตรงนี้ ทำให้หลุดจาก switch ถ้าชนกับ Object ทางขวา
					}
				}
				// --- รีเซ็ตตำแหน่ง solidArea กลับไปค่าเริ่มต้น ---
				// สำคัญมาก! ต้องรีเซ็ตกลับไป เพื่อไม่ให้ solidArea ของ Entity/Object มันค้างตำแหน่งผิด
				entity.solidArea.x = entity.solidAreaDefaultX;
				entity.solidArea.y = entity.solidAreaDefaultY;
				gp.obj[i].solidArea.x = gp.obj[i].solidAreaDefaultX;
				gp.obj[i].solidArea.y = gp.obj[i].solidAreaDefaultY;
			}
		}		
		return index; // คืนค่า index ของ Object ที่ชน (ถ้ามี) หรือ 999 (ถ้าไม่ชน)
	}
}
