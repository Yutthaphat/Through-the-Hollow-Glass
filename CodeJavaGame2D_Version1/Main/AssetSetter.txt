// อยู่ในแพ็กเกจ JavaLearn นะ อันนี้สำหรับจัดการพวก "ของ" ในเกม
package JavaLearn;

// ดึงคลาส Object ต่างๆ ที่เราสร้างไว้มาใช้
import Object.OBJ_Boots;
import Object.OBJ_Chest;
import Object.OBJ_Door;
import Object.OBJ_Key;

// คลาส AssetSetter นี้มีหน้าที่เหมือนเป็นคนจัดฉากเกม วางของต่างๆ ลงไปใน World Map
public class AssetSetter {
	
	GamePanel gp; // อ้างอิงถึง GamePanel เพื่อที่เราจะไปวางของใน Array ของ GamePanel ได้
	
	// Constructor ของ AssetSetter
	public AssetSetter(GamePanel gp) {
		this.gp = gp;
	}
	
	// เมธอด setObject() นี้แหละคือที่ที่เราจะสร้าง Object แล้วก็กำหนดตำแหน่งให้มัน
	public void setObject() {
		
		// สร้างกุญแจ 3 ดอก แล้วก็บอกว่ามันอยู่ตรงไหนของ World Map (ใช้ worldX, worldY)
		gp.obj[0] = new OBJ_Key();
		gp.obj[0].worldX = 23 * gp.tileSize;
		gp.obj[0].worldY = 7 * gp.tileSize;
		
		gp.obj[1] = new OBJ_Key();
		gp.obj[1].worldX = 23 * gp.tileSize;
		gp.obj[1].worldY = 40 * gp.tileSize;
		
		gp.obj[2] = new OBJ_Key();
		gp.obj[2].worldX = 37 * gp.tileSize;
		gp.obj[2].worldY = 7 * gp.tileSize;
		
		// สร้างประตู 3 บาน กำหนดตำแหน่งให้มันด้วย
		gp.obj[3] = new OBJ_Door();
		gp.obj[3].worldX = 10 * gp.tileSize;
		gp.obj[3].worldY = 11 * gp.tileSize;
		
		gp.obj[4] = new OBJ_Door();
		gp.obj[4].worldX = 8 * gp.tileSize;
		gp.obj[4].worldY = 28 * gp.tileSize;
		
		gp.obj[5] = new OBJ_Door();
		gp.obj[5].worldX = 12 * gp.tileSize;
		gp.obj[5].worldY = 22 * gp.tileSize;
		
		// สร้างหีบสมบัติ 1 กล่อง
		gp.obj[6] = new OBJ_Chest();
		gp.obj[6].worldX = 10 * gp.tileSize;
		gp.obj[6].worldY = 7 * gp.tileSize;
		
		// สร้างรองเท้า 1 คู่
		gp.obj[7] = new OBJ_Boots();
		gp.obj[7].worldX = 37 * gp.tileSize;
		gp.obj[7].worldY = 42 * gp.tileSize;
	}
}
