package JavaLearn;

import javax.swing.JFrame; // นำเข้าคลาส JFrame สำหรับสร้างหน้าต่าง GUI

public class Main {
	public static void main(String[] args) {
		
		// สร้างวัตถุ JFrame เพื่อเป็นหน้าต่างหลักของเกม
		JFrame window = new JFrame();
		
		// ตั้งค่าการทำงานเมื่อกดปุ่มปิด (กากบาท) บนหน้าต่าง: ปิดโปรแกรมเมื่อปิดหน้าต่าง
		window.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);  
		
		// ตั้งค่าไม่ให้ปรับขนาดหน้าต่างได้
		window.setResizable(false);  
		
		// ตั้งชื่อหัวข้อของหน้าต่าง
		window.setTitle("2D Adventure");  
		
		// สร้างวัตถุ GamePanel ซึ่งเป็นส่วนที่วาดและจัดการเกมทั้งหมด
		GamePanel gamePanel = new GamePanel();
		
		// เพิ่ม GamePanel เข้าไปในหน้าต่าง
		window.add(gamePanel);
		
		// ปรับขนาดหน้าต่างให้พอดีกับส่วนประกอบภายใน (GamePanel)
		window.pack();
		
		// ตั้งตำแหน่งหน้าต่างให้อยู่กึ่งกลางของหน้าจอ (null หมายถึงอยู่ตรงกลาง)
		window.setLocationRelativeTo(null);  
		
		// ทำให้หน้าต่างแสดงผลบนหน้าจอ
		window.setVisible(true); 
		
		// เริ่ม Game Loop (Thread) ของ GamePanel เพื่อให้เกมทำงาน
		gamePanel.startGameThread();
	}
}
