package JavaLearn;

import java.awt.Color;         // สำหรับกำหนดสี
import java.awt.Dimension;     // สำหรับกำหนดขนาด
import java.awt.Graphics;      // คลาสพื้นฐานสำหรับวาดภาพ
import java.awt.Graphics2D;    // คลาสสำหรับวาดภาพ 2 มิติขั้นสูง

import javax.swing.JPanel;     // คลาสสำหรับสร้าง Panel ใน GUI

import Entity.Player;          // นำเข้าคลาส Player
import Tile.TileManager;       // นำเข้าคลาส TileManager

public class GamePanel extends JPanel implements Runnable{ // GamePanel เป็น JPanel และ implements Runnable เพื่อให้สามารถรันใน Thread ได้
	
	// ขนาดเริ่มต้นของ Tile (กระเบื้อง) ในเกม: 16x16 พิกเซล
	final int originalTileSize = 16;  
	// ตัวคูณขนาด เพื่อขยาย Tile ให้ใหญ่ขึ้น (16 * 3 = 48 พิกเซล)
	final int scale = 3; 
	// ขนาด Tile สุดท้ายที่จะแสดงบนหน้าจอ: 48x48 พิกเซล
	public final int tileSize = originalTileSize * scale;  
	
	// จำนวน Tile สูงสุดในแนวนอนของหน้าจอ (16 Tile)
	public final int maxScreenCol = 16;  
	// จำนวน Tile สูงสุดในแนวตั้งของหน้าจอ (12 Tile)
	public final int maxScreenRow = 12;  
	
	// ความกว้างของหน้าจอเกมเป็นพิกเซล (48 * 16 = 768 พิกเซล)
	public final int screenWidth = tileSize * maxScreenCol;  
	// ความสูงของหน้าจอเกมเป็นพิกเซล (48 * 12 = 576 พิกเซล)
	public final int screenHeight = tileSize * maxScreenRow;  

	// กำหนด Frame Per Second (FPS) ของเกม
	int FPS = 60;

	// สร้างวัตถุ TileManager สำหรับจัดการแผนที่
	TileManager tileM = new TileManager(this);
	// สร้างวัตถุ KeyHandler สำหรับจัดการการกดปุ่ม
	KeyHandler keyH = new KeyHandler();
	// Thread สำหรับ Game Loop ทำให้โปรแกรมรันได้จนกว่าเราจะหยุด
	Thread gameThread;  
	// สร้างวัตถุ Player สำหรับตัวละครผู้เล่น
	Player player = new Player(this,keyH);

	// กำหนดตำแหน่งเริ่มต้นของผู้เล่น (ค่าเริ่มต้นที่ไม่ได้ใช้ใน Player Class)
	//int playerX = 100; 
	//int playerY = 100;
	// กำหนดความเร็วเริ่มต้นของผู้เล่น (ค่าเริ่มต้นที่ไม่ได้ใช้ใน Player Class)
	//int playerSpeed = 4;  

	public GamePanel () {
		// กำหนดขนาดที่ต้องการของ JPanel นี้ ให้เท่ากับขนาดหน้าจอเกมที่คำนวณไว้
		this.setPreferredSize(new Dimension(screenWidth, screenHeight));  
		// ตั้งค่าสีพื้นหลังของ Panel เป็นสีดำ
		this.setBackground(Color.black);   
		// เปิดใช้งาน Double Buffering เพื่อลดการกระพริบของภาพ ทำให้ภาพลื่นไหลต่อเนื่อง
		// คือการวาดทุกอย่างลงในหน่วยความจำก่อน แล้วค่อยแสดงผลทีเดียวบนหน้าจอจริง
		this.setDoubleBuffered(true);  
		// เพิ่ม KeyListener เพื่อให้ Panel รับอินพุตจากคีย์บอร์ด
		this.addKeyListener(keyH);  
		// กำหนดให้ Panel สามารถรับโฟกัสได้ เพื่อให้รับอินพุตจากคีย์บอร์ด
		this.setFocusable(true); 

	}
	
	// เมธอดสำหรับเริ่ม Game Loop Thread
	public void startGameThread(){
		gameThread = new Thread(this); // สร้าง Thread ใหม่โดยใช้ GamePanel เป็น Runnable
		gameThread.start(); // เริ่ม Thread
	}

	@Override
	public void run() {	
		// คำนวณช่วงเวลาที่ควรจะวาดแต่ละเฟรม (เป็นหน่วยนาโนวินาที)
		// เช่น FPS = 60 --> 1,000,000,000 ns / 60 = 16,666,666 ns ต่อเฟรม
		double drawInterval = 1000000000/FPS;  
		// ตัวแปรสำหรับสะสมเวลา เพื่อดูว่าถึงเวลาวาดเฟรมถัดไปหรือยัง
		double delta = 0;   
		// บันทึกเวลาล่าสุดที่ตรวจสอบ (เป็นนาโนวินาที)
		long lastTime = System.nanoTime();  
		// ตัวแปรสำหรับเก็บเวลาปัจจุบัน
		long currentTime;
		// ตัวแปรสำหรับจับเวลาครบ 1 วินาที เพื่อแสดง FPS
		long timer = 0;   
		// ตัวแปรสำหรับนับจำนวนเฟรมที่วาดไปใน 1 วินาที
		int drawCount = 0;   
		
		// Game Loop: วนลูปทำงานตราบเท่าที่ gameThread ยังไม่เป็น null (เกมยังไม่หยุด)
		while(gameThread != null){    
			currentTime = System.nanoTime();   
			// คำนวณเวลาที่ผ่านไปตั้งแต่ lastTime และนำมาสะสมใน delta
			// delta จะเพิ่มขึ้นเรื่อยๆ และเมื่อ >= 1 หมายถึงถึงเวลาวาด 1 เฟรมแล้ว
			delta += (currentTime - lastTime) / drawInterval;  
			// คำนวณเวลาที่ผ่านไปเพื่อนำไปใช้ในการจับเวลา FPS
			timer += (currentTime - lastTime);    
			// อัปเดต lastTime เป็นเวลาปัจจุบันสำหรับรอบถัดไป
			lastTime = currentTime;

			// ถ้า delta >= 1 หมายถึงถึงเวลาที่จะอัปเดตและวาด 1 เฟรมแล้ว
			if(delta >= 1){
				update();      // เรียกเมธอด update() เพื่ออัปเดตสถานะเกม
				repaint();     // เรียกเมธอด repaint() เพื่อวาดหน้าจอใหม่ (จะไปเรียก paintComponent())
				delta--;       // ลดค่า delta ลง 1 เพื่อเตรียมพร้อมสำหรับเฟรมถัดไป
				drawCount++;   // เพิ่มจำนวนเฟรมที่วาดไป
			}

			// ถ้า timer ครบ 1 วินาที (1,000,000,000 นาโนวินาที)
			if(timer >= 1000000000){  
				// แสดงค่า FPS จริงๆ ที่ทำได้ในวินาทีนั้นๆ ออกทาง Console
				System.out.println("FPS" + drawCount);
				drawCount = 0; // รีเซ็ตจำนวนเฟรม
				timer = 0;     // รีเซ็ตตัวจับเวลา
			}
		}
	}

	// เมธอดสำหรับอัปเดตสถานะของเกม
	public void update() {
		player.update(); // เรียกเมธอด update ของผู้เล่นเพื่ออัปเดตตำแหน่งและการเคลื่อนไหว
	}
	
	// เมธอดสำหรับวาดส่วนประกอบต่างๆ บน JPanel (ถูกเรียกโดย repaint())
	public void paintComponent(Graphics g) {
		super.paintComponent(g); // เรียกเมธอด paintComponent ของ JPanel เพื่อวาดพื้นหลัง
		
		Graphics2D g2 = (Graphics2D)g; // แปลง Graphics เป็น Graphics2D เพื่อใช้งานฟังก์ชันการวาด 2 มิติขั้นสูง
		
		tileM.draw(g2); // วาดแผนที่ (Tiles)
		
		player.draw(g2); // วาดตัวละครผู้เล่น
		
		g2.dispose();  // ปล่อยทรัพยากร Graphics2D เมื่อวาดเสร็จ เพื่อประสิทธิภาพที่ดี (ป้องกัน memory leak)
	}
}
