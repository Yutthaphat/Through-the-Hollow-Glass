package JavaLearn;

import java.awt.event.KeyEvent;    // สำหรับเหตุการณ์ที่เกี่ยวกับคีย์บอร์ด
import java.awt.event.KeyListener; // อินเทอร์เฟซสำหรับรับการกดปุ่ม

public class KeyHandler implements KeyListener{  // KeyHandler implements KeyListener เพื่อรับค่าอินพุตจากคีย์บอร์ด
	
	// ตัวแปร boolean เพื่อบอกว่าปุ่มทิศทางแต่ละปุ่มถูกกดอยู่หรือไม่
	public boolean upPressed, downPressed, leftPressed, rightPressed;
	
	@Override
	public void keyTyped(KeyEvent e) {
		// เมธอดนี้จะถูกเรียกเมื่อมีการพิมพ์อักขระ (ไม่ค่อยได้ใช้ในเกมส่วนใหญ่)
	}
	
	@Override
	public void keyPressed(KeyEvent e) {
		// เมธอดนี้จะถูกเรียกเมื่อมีการกดปุ่มลง
		int code = e.getKeyCode(); // รับโค้ดของปุ่มที่ถูกกด
		
		if (code == KeyEvent.VK_W){  // ถ้าปุ่มที่กดคือ 'W' (สำหรับเดินขึ้น)
			upPressed = true; // ตั้งค่า upPressed เป็น true
		}
		if (code == KeyEvent.VK_S){    // ถ้าปุ่มที่กดคือ 'S' (สำหรับเดินลง)
			downPressed = true; // ตั้งค่า downPressed เป็น true
		}	
		if (code == KeyEvent.VK_A){   // ถ้าปุ่มที่กดคือ 'A' (สำหรับเดินซ้าย)
			leftPressed = true; // ตั้งค่า leftPressed เป็น true
		}		
		if (code == KeyEvent.VK_D){   // ถ้าปุ่มที่กดคือ 'D' (สำหรับเดินขวา)
			rightPressed = true; // ตั้งค่า rightPressed เป็น true
		}	
	}
	
	@Override
	public void keyReleased(KeyEvent e) {
		// เมธอดนี้จะถูกเรียกเมื่อมีการปล่อยปุ่ม
		int code = e.getKeyCode(); // รับโค้ดของปุ่มที่ถูกปล่อย
		
		if (code == KeyEvent.VK_W){  // ถ้าปุ่มที่ปล่อยคือ 'W' (ถ้าปล่อย W ก็ไม่เดินขึ้นแล้ว)
			upPressed = false; // ตั้งค่า upPressed เป็น false
		}
		if (code == KeyEvent.VK_S){    // ถ้าปุ่มที่ปล่อยคือ 'S' (ถ้าปล่อย S ก็ไม่เดินลงแล้ว)
			downPressed = false; // ตั้งค่า downPressed เป็น false
		}	
		if (code == KeyEvent.VK_A){   // ถ้าปุ่มที่ปล่อยคือ 'A' (ถ้าปล่อย A ก็ไม่เดินซ้ายแล้ว)
			leftPressed = false; // ตั้งค่า leftPressed เป็น false
		}		
		if (code == KeyEvent.VK_D){   // ถ้าปุ่มที่ปล่อยคือ 'D' (ถ้าปล่อย D ก็ไม่เดินขวาแล้ว)
			rightPressed = false; // ตั้งค่า rightPressed เป็น false
		}
	}
}
