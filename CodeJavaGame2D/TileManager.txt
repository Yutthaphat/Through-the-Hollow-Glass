package Tile;

import java.awt.Graphics2D;        // สำหรับวาดภาพ 2 มิติขั้นสูง
import java.io.BufferedReader;      // สำหรับอ่านข้อความจาก Stream (มีประสิทธิภาพดีกว่า InputStreamReader ตรงที่อ่านเป็น buffer)
import java.io.IOException;         // สำหรับจัดการข้อผิดพลาดเกี่ยวกับการ I/O
import java.io.InputStream;         // สำหรับอ่านไบต์จาก Stream
import java.io.InputStreamReader;   // สำหรับแปลงไบต์ Stream เป็น Character Stream

import javax.imageio.ImageIO;       // สำหรับอ่านไฟล์รูปภาพ

import JavaLearn.GamePanel;         // นำเข้าคลาส GamePanel

public class TileManager {
	
	GamePanel gp;        // อ้างอิงถึง GamePanel เพื่อเข้าถึงข้อมูลของเกม
	Tile[] tile;         // อาร์เรย์สำหรับเก็บวัตถุ Tile แต่ละประเภท (เช่น หญ้า, กำแพง, น้ำ)
	int mapTileNum[][];  // อาร์เรย์ 2 มิติสำหรับเก็บหมายเลข Tile ที่ระบุว่าในแต่ละตำแหน่งของแผนที่เป็น Tile ประเภทใด
	
	// Constructor สำหรับ TileManager
	public TileManager(GamePanel gp) {
		
		this.gp = gp; // กำหนด GamePanel ที่รับมาให้กับตัวแปร gp
		
		tile = new Tile[10]; // สร้างอาร์เรย์ Tile ขนาด 10 ช่อง (สำหรับเก็บ Tile ได้ 10 ประเภท)
		// สร้างอาร์เรย์ 2 มิติสำหรับแผนที่ โดยมีขนาดเท่ากับ maxScreenCol และ maxScreenRow ของ GamePanel
		mapTileNum = new int[gp.maxScreenCol][gp.maxScreenRow];
		
		getTileImage();        // เรียกเมธอดเพื่อโหลดรูปภาพของ Tile ต่างๆ
		loadMap("/maps/map01.txt"); // เรียกเมธอดเพื่อโหลดข้อมูลแผนที่จากไฟล์
	}
	
	// เมธอดสำหรับโหลดรูปภาพของ Tile ประเภทต่างๆ
	public void getTileImage() {
		
		try {
			// สร้างวัตถุ Tile และโหลดรูปภาพสำหรับแต่ละประเภท
			
			// Tile ประเภท 0: หญ้า
			tile[0] = new Tile();
			tile[0].image = ImageIO.read(getClass().getResourceAsStream("/tiles/Grass.png"));
			
			// Tile ประเภท 1: กำแพง
			tile[1] = new Tile();
			tile[1].image = ImageIO.read(getClass().getResourceAsStream("/tiles/Wall.png"));

			// Tile ประเภท 2: น้ำ
			tile[2] = new Tile();
			tile[2].image = ImageIO.read(getClass().getResourceAsStream("/tiles/Water.png"));
			
		}catch(IOException e) {
			// หากเกิดข้อผิดพลาดในการโหลดรูปภาพ จะแสดง Stack Trace
			e.printStackTrace();
		}
	}
	
	// เมธอดสำหรับโหลดข้อมูลแผนที่จากไฟล์ .txt
	public void loadMap(String filePath) {
		
		try {
			// รับ InputStream จากไฟล์แผนที่ใน Resources
			InputStream is = getClass().getResourceAsStream(filePath);
			// สร้าง BufferedReader เพื่ออ่านข้อความจาก InputStream ได้อย่างมีประสิทธิภาพ
			BufferedReader br = new BufferedReader(new InputStreamReader(is));
			
			int col = 0; // ตัวนับคอลัมน์
			int row = 0; // ตัวนับแถว
			
			// วนลูปจนกว่าจะอ่านครบทุกคอลัมน์และทุกแถวของแผนที่
			while(col < gp.maxScreenCol && row < gp.maxScreenRow) {
				
				String line = br.readLine(); // อ่านข้อมูลแผนที่ทีละบรรทัด
				
				// วนลูปเพื่ออ่านค่า Tile ในแต่ละคอลัมน์ของบรรทัดปัจจุบัน
				while(col < gp.maxScreenCol) {
					
					// แยกตัวเลขในบรรทัดโดยใช้ช่องว่างเป็นตัวคั่น
					String numbers[] = line.split(" ");
					
					// แปลง String เป็น Integer
					int num = Integer.parseInt(numbers[col]);
					
					// เก็บหมายเลข Tile ลงในอาร์เรย์ mapTileNum
					mapTileNum[col][row] = num;
					col++; // เลื่อนไปคอลัมน์ถัดไป
				}
				// ถ้าอ่านครบทุกคอลัมน์ในบรรทัดปัจจุบันแล้ว
				if(col == gp.maxScreenCol) {
					col = 0; // รีเซ็ตคอลัมน์เป็น 0
					row++;   // เลื่อนไปแถวถัดไป
				}
			}
			br.close(); // ปิด BufferedReader เมื่ออ่านเสร็จ
			
		}catch(Exception e) {
			// หากเกิดข้อผิดพลาดใดๆ ในระหว่างการโหลดแผนที่ (เช่น ไฟล์ไม่พบ, รูปแบบผิดพลาด)
			e.printStackTrace(); // แสดง Stack Trace ของข้อผิดพลาด
		}
	}
	
	// เมธอดสำหรับวาดแผนที่บนหน้าจอ
	public void draw(Graphics2D g2) {
		
		/*
		// โค้ดที่คอมเมนต์ไว้ด้านล่างเป็นตัวอย่างการวาด Tile แบบ Hardcode (ไม่แนะนำสำหรับแผนที่ใหญ่)
		// g2.drawImage(tile[1].image, 0, 0, gp.tileSize, gp.tileSize, null);
		// ... (โค้ดเดิม)
		*/
		
		int col = 0; // ตัวนับคอลัมน์
		int row = 0; // ตัวนับแถว
		int x = 0;   // ตำแหน่ง X สำหรับวาด Tile (เป็นพิกเซล)
		int y = 0;   // ตำแหน่ง Y สำหรับวาด Tile (เป็นพิกเซล)
		
		// วนลูปเพื่อวาด Tile ทุก Tile บนหน้าจอ
		while(col < gp.maxScreenCol && row < gp.maxScreenRow) {
			
			// ดึงหมายเลข Tile จากอาร์เรย์ mapTileNum
			int tileNum = mapTileNum[col][row];
			
			// วาดรูปภาพ Tile ที่ตำแหน่ง (x, y) ด้วยขนาด tileSize
			g2.drawImage(tile[tileNum].image, x, y, gp.tileSize, gp.tileSize, null);
			col++;         // เลื่อนไปคอลัมน์ถัดไป
			x += gp.tileSize; // เพิ่มตำแหน่ง X สำหรับ Tile ถัดไป
			
			// ถ้าวาดครบทุกคอลัมน์ในแถวปัจจุบันแล้ว
			if(col == gp.maxScreenCol) {
				col = 0;        // รีเซ็ตคอลัมน์เป็น 0
				x = 0;          // รีเซ็ตตำแหน่ง X เป็น 0
				row++;          // เลื่อนไปแถวถัดไป
				y += gp.tileSize; // เพิ่มตำแหน่ง Y สำหรับแถวถัดไป
			}
		}
	}
}
